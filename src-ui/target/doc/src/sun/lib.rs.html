3//! The `sun` crate is a library for calculating the position of the sun.
4//! It is a port of the JavaScript library
5//! [suncalc](https://github.com/mourner/suncalc).
6//!
7//! # Example
8//!
9//! ```
10//! extern crate sun;
11//!
12//! pub fn main() {
13//!   let unixtime = 1362441600000;
14//!   let lat = 48.0;
15//!   let lon = 9.0;
16//!   let pos = sun::pos(unixtime,lat,lon);
17//!   let az  = pos.azimuth.to_degrees();
18//!   let alt = pos.altitude.to_degrees();
19//!   println!("The position of the sun is {}/{}", az, alt);
20//! }
21//! ```
22
23use std::f64::consts::PI;
24
25// date/time constants and conversions
26
27const MILLISECONDS_PER_DAY  : u32 = 1000 * 60 * 60 * 24;
28const J1970                 : u32 = 2440588;
29const J2000                 : u32 = 2451545;
30const TO_RAD                : f64 = PI / 180.0;
31const OBLIQUITY_OF_EARTH    : f64 = 23.4397  * TO_RAD;
32const PERIHELION_OF_EARTH   : f64 = 102.9372 * TO_RAD;
33
34/// Holds the [azimuth](https://en.wikipedia.org/wiki/Azimuth)
35/// and [altitude](https://en.wikipedia.org/wiki/Horizontal_coordinate_system)
36/// angles of the sun position.
37#[derive(Debug)]
38pub struct Position {
39  pub azimuth   : f64,
40  pub altitude  : f64
41}
42
43fn to_julian(unixtime_in_ms: i64) -> f64 {
44  unixtime_in_ms as f64 /
45  (MILLISECONDS_PER_DAY as f64) - 0.5 + J1970 as f64
46}
47
48#[test]
49fn test_to_julian(){
50  // 1. Jan. 2015
51  assert_eq!(2457054.5, to_julian(1422748800000));
52}
53
54fn to_days(unixtime_in_ms: i64) -> f64 {
55  to_julian(unixtime_in_ms) - J2000 as f64
56}
57
58#[test]
59fn test_to_days(){
60  // 1. Jan. 2015
61  assert_eq!(5509.5, to_days(1422748800000));
62}

64// general calculations for position
65
66fn right_ascension(l:f64, b:f64) -> f64 {
67  (
68    l.sin() * OBLIQUITY_OF_EARTH.cos() -
69    b.tan() * OBLIQUITY_OF_EARTH.sin()
70  )
71  .atan2(l.cos())
72}
73
74fn declination(l:f64, b:f64) -> f64 {
75  (
76    b.sin() * OBLIQUITY_OF_EARTH.cos() +
77    b.cos() * OBLIQUITY_OF_EARTH.sin() * l.sin()
78  )
79  .asin()
80}
81
82fn azimuth(h:f64, phi:f64, dec:f64) -> f64  {
83  h.sin()
84  .atan2(
85    h.cos()   * phi.sin() -
86    dec.tan() * phi.cos()
87  ) + PI
88}
89
90fn altitude(h:f64, phi:f64, dec:f64) -> f64 {
91  (
92    phi.sin() * dec.sin() +
93    phi.cos() * dec.cos() * h.cos()
94  )
95  .asin()
96}
97
98fn sidereal_time(d:f64, lw:f64) -> f64 {
99  (280.16 + 360.9856235 * d).to_radians() - lw
100}
101
102// general sun calculations
103
104fn solar_mean_anomaly(d:f64) -> f64 {
105  (357.5291 + 0.98560028 * d).to_radians()
106}
107
108fn equation_of_center(m:f64) -> f64 {
109  (1.9148 * (1.0 * m).sin() +
110   0.02   * (2.0 * m).sin() +
111   0.0003 * (3.0 * m).sin()
112  ).to_radians()
113}
114
115fn ecliptic_longitude(m:f64) -> f64 {
116  m + equation_of_center(m) + PERIHELION_OF_EARTH + PI
117}

119/// Calculates the sun position for a given date and latitude/longitude.
120/// The angles are calculated as [radians](https://en.wikipedia.org/wiki/Radian).
121///
122/// * `unixtime`  - [unix time](https://en.wikipedia.org/wiki/Unix_time) in milliseconds.
123/// * `lat`       - [latitude](https://en.wikipedia.org/wiki/Latitude) in degrees.
124/// * `lon`       - [longitude](https://en.wikipedia.org/wiki/Longitude) in degrees.
125/// calculates the sun position for a given date and latitude/longitude
126pub fn pos(unixtime_in_ms: i64, lat: f64, lon: f64) -> Position {
127
128  let lw  = -lon.to_radians();
129  let phi = lat.to_radians();
130  let d   = to_days(unixtime_in_ms);
131  let m   = solar_mean_anomaly(d);
132  let l   = ecliptic_longitude(m);
133  let dec = declination(l, 0.0);
134  let ra  = right_ascension(l, 0.0);
135  let h   = sidereal_time(d, lw) - ra;
136
137  Position {
138    azimuth  :  azimuth(h, phi, dec),
139    altitude : altitude(h, phi, dec)
140  }
141}
142
143#[test]
144fn test_pos(){
145  // 2013-03-05 UTC
146  let date = 1362441600000;
147  let pos = pos(date, 50.5, 30.5);
148  assert_eq!(0.6412750628729547, pos.azimuth);
149  assert_eq!(-0.7000406838781611, pos.altitude);
150}
